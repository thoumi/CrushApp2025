    📝 Commandes Essentielles - CrushApp Microservices¶

    🚀 Démarrage Rapide¶

    Configuration Initiale¶

    # Créer le fichier .env
Copy-Item .env.example .env

# Éditer avec vos clés Cloudinary
notepad .env

    Lancement¶

    # Démarrer tous les services
docker-compose up -d

# Suivre les logs en temps réel
docker-compose logs -f

# Logs d'un service spécifique
docker-compose logs -f core-api
docker-compose logs -f chatbot-service
docker-compose logs -f media-service

    🔍 Vérification¶

    Health Checks¶

    # Gateway
curl http://localhost:5000/health

# Core API
curl http://localhost:5001/health

# Chatbot Service
curl http://localhost:5002/health

# Media Service
curl http://localhost:5003/health

    Status Services¶

    # Voir tous les services
docker-compose ps

# Statistiques en temps réel
docker stats

    🧪 Tests¶

    Créer un compte¶

    curl -X POST http://localhost:5000/api/account/register `
  -H "Content-Type: application/json" `
  -d '{
    "username": "testuser",
    "password": "Pa$$w0rd",
    "displayName": "Test User",
    "dateOfBirth": "1990-01-01",
    "city": "Paris",
    "country": "France",
    "gender": "male"
  }'

    Se connecter¶

    curl -X POST http://localhost:5000/api/account/login `
  -H "Content-Type: application/json" `
  -d '{
    "username": "testuser",
    "password": "Pa$$w0rd"
  }'

# Récupérer le token de la réponse

    Tester le Chatbot¶

    # Remplacer YOUR_TOKEN
curl -X POST http://localhost:5000/api/chatbot/chat `
  -H "Authorization: Bearer YOUR_TOKEN" `
  -H "Content-Type: application/json" `
  -d '{"message": "Bonjour !"}'

    Uploader une photo¶

    # Remplacer YOUR_TOKEN et le chemin
curl -X POST http://localhost:5000/api/photos `
  -H "Authorization: Bearer YOUR_TOKEN" `
  -F "file=@C:\path\to\photo.jpg"

    🛠️ Gestion¶

    Redémarrer¶

    # Tous les services
docker-compose restart

# Un service spécifique
docker-compose restart core-api
docker-compose restart chatbot-service
docker-compose restart media-service

    Arrêter¶

    # Arrêter tous les services
docker-compose stop

# Arrêter et supprimer
docker-compose down

# Supprimer aussi les volumes
docker-compose down -v

    Reconstruire¶

    # Reconstruire tous les services
docker-compose up -d --build

# Reconstruire un service spécifique
docker-compose up -d --build core-api

@Level = 'Information'

    🗄️ Base de Données¶

    Accéder à SQL Server¶

    # Via Docker
docker-compose exec sqlserver /opt/mssql-tools/bin/sqlcmd `
  -S localhost -U sa -P "REDACTED_DB_PASSWORD"

# Lister les bases
SELECT name FROM sys.databases;
GO

# Utiliser une base
USE CrushAppDb;
GO

# Lister les tables
SELECT * FROM INFORMATION_SCHEMA.TABLES;
GO

    📦 RabbitMQ¶

    CLI¶

    # Lister les queues
docker-compose exec rabbitmq rabbitmqctl list_queues

# Lister les connections
docker-compose exec rabbitmq rabbitmqctl list_connections

# Lister les exchanges
docker-compose exec rabbitmq rabbitmqctl list_exchanges

- photo.uploaded
- photo.approved
- photo.rejected

# Voir les messages
Queues → [nom_queue] → Get messages

    🧹 Nettoyage¶

    Nettoyer tout¶

    # Arrêter et supprimer containers + volumes
docker-compose down -v

# Supprimer images inutilisées
docker image prune -a

# Nettoyer complètement Docker
docker system prune -a --volumes

    Nettoyer Build Cache¶

    # Nettoyer cache Docker
docker builder prune -a

    🔧 Troubleshooting¶

    Service ne démarre pas¶

    # Voir les logs d'erreur
docker-compose logs SERVICE_NAME

# Redémarrer avec rebuild
docker-compose up -d --build SERVICE_NAME

# Vérifier la configuration
docker-compose config

    Port déjà utilisé¶

    # Trouver quel process utilise le port
netstat -ano | findstr :5000

# Tuer le process
taskkill /PID PROCESS_ID /F

    RabbitMQ ne démarre pas¶

    # Vérifier les logs
docker-compose logs rabbitmq

# Supprimer les données et redémarrer
docker-compose down -v
docker-compose up -d rabbitmq

    Base de données non créée¶

    # Vérifier migrations
docker-compose logs core-api | Select-String "Migration"

# Forcer les migrations
docker-compose exec core-api dotnet ef database update

    🎯 Workflows Courants¶

    Développement¶

    # 1. Modifier le code
# 2. Reconstruire le service
docker-compose up -d --build SERVICE_NAME

# 3. Voir les logs
docker-compose logs -f SERVICE_NAME

    Debug¶

    # 1. Arrêter le service
docker-compose stop SERVICE_NAME

# 2. Voir les logs complets
docker-compose logs SERVICE_NAME > logs.txt

# 3. Redémarrer
docker-compose up -d SERVICE_NAME

    Production Deploy¶

    # 1. Build
docker-compose build

# 2. Test local
docker-compose up -d

# 3. Vérifier
.\verify-migration-simple.ps1

# 4. Tag & Push (si registry)
docker-compose push
